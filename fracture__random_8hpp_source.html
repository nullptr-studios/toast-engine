42inline auto sphereDirection(float u, float v) noexcept -> glm::vec3 {
43 float z = (u * 2.0f) - 1.0f;
44 float ring = std::sqrt(glm::max(1.0f - (z * z), 0.0f));
45 float phi = v * 2.0f * std::numbers::pi_v<float>;
46 return {ring * std::cos(phi), ring * std::sin(phi), z};
47}

58inline auto fractureSeed(BodyID source, ShapeID shape, glm::vec3 local_point, uint64_t tick) noexcept -> uint64_t {
59 // calculates the seed of one hit
60 uint64_t seed = 0;
61 toast::hashCombine(seed, (static_cast<uint64_t>(source.generation) << 32) | source.slot);
62 toast::hashCombine(seed, (static_cast<uint64_t>(shape.generation) << 32) | source.slot);
63 seed = hashCell(seed, glm::ivec3(glm::floor(local_point / voxel::k_voxel_size)));
64 toast::hashCombine(seed, tick);

80inline auto valueNoise3(uint64_t seed, glm::vec3 p) noexcept -> float {
81 // trilinear of the random
82 glm::vec3 base = glm::floor(p);
83 glm::ivec3 cell = {base};
84 glm::vec3 t = p - base;
85 glm::vec3 w = t * t * (3.0f - (2.0f * t));
86
87#define LARP(a, b, c) std::lerp(a, b, c)
88
89 auto at = [&](int x, int y, int z) { return hash01(seed, cell + glm::ivec3 {x, y, z}); };
90 float x00 = LARP(at(0, 0, 0), at(1, 0, 0), w.x);
91 float x01 = LARP(at(0, 1, 0), at(1, 1, 0), w.x);
92 float x10 = LARP(at(0, 0, 1), at(1, 0, 1), w.x);
93 float x11 = LARP(at(0, 1, 1), at(1, 1, 1), w.x);
94 return LARP(LARP(x00, x10, w.y), LARP(x01, x11, w.y), w.z);
