60 Hardness max_hardness

62 ZoneScopedN("physics::PlanShatter");
63 ShatterPlan out;
64 out.band_outer = craterMaxRadius(crater);

66 const FractureValues& values = crater.profile;
67 if (not crater.breaks || not(crater.shatter_energy > 0.0f) || not patternShatters(crater.pattern) ||
68 not(crater.radius > 0.0f) || materials.materials.empty()) {
69 out.stop = "preconditions";
70 return out;
71 }
72
73 const float crater_volume = craterVolume(crater);
74 out.density = crater.shatter_energy / crater_volume;
75 if (out.density < values.shatter_threshold * crater.toughness) {
76 out.stop = "energy density below the threshold of the material";
77 return out;
78 }

80 const float threshold = crackThickness();
81 const float min_width = 2.0f * threshold;
82 const float max_width = std::max(tunables().fracture_max_shatter_distance, min_width);
83
84 const float cell_size = std::max(values.shatter_cell_size, threshold);
85 const float length = glm::length(crater.end - crater.start);
86 const float base_volume = capsuleVolume(crater.radius, length);
87 const float fill = std::min(1.0f, 3.0f * threshold / cell_size);
88 const auto band_volume_of = [&](float w) { return capsuleVolume(crater.radius + w, length) - base_volume; };
89 const auto width_for = [&](float mean_toughness) {
90 // widest band for the shatter the enerby pays for
91 const auto cost_of = [&](float w) { return mean_toughness * values.shatter_crack_cost * fill * band_volume_of(w); };
92 if (cost_of(max_width) <= crater.shatter_energy) {
93 return max_width;
94 }
95 if (not(cost_of(min_width) < crater.shatter_energy)) {
96 return min_width;
97 }
98 float low = min_width;
99 float high = max_width;
100 for (int i = 0; i < 24; ++i) {
101 const float middle = 0.5f * (low + high);
102 (cost_of(middle) <= crater.shatter_energy ? low : high) = middle;
103 }
104 return low;
105 };

267 struct Owner {
268 size_t nearest = 0;
269 size_t second = 0;
270 float gap = 0.0f; // d2 - d1
271 };
272
273 std::vector<Owner> owners(band.size());
274 std::vector<uint32_t> cell_voxels(seeds.size(), 0);
275 std::vector<glm::ivec3> cell_low(seeds.size(), glm::ivec3 {std::numeric_limits<int32_t>::max()});
276 std::vector<glm::ivec3> cell_high(seeds.size(), glm::ivec3 {std::numeric_limits<int32_t>::min()});
277 for (size_t b = 0; b < band.size(); ++b) {
278 float d1 = std::numeric_limits<float>::max();
279 float d2 = std::numeric_limits<float>::max();
280 Owner& owner = owners[b];
281 for (size_t i = 0; i < seeds.size(); ++i) {
282 const float d = metric_squared(band[b].center - seeds[i]);
283 if (d < d1) {
284 d2 = d1;
285 owner.second = owner.nearest;
286 d1 = d;
287 owner.nearest = i;
288 } else if (d < d2) {
289 d2 = d;
290 owner.second = i;
291 }
292 }
293 owner.gap = std::sqrt(d2) - std::sqrt(d1);
294 ++cell_voxels[owner.nearest];
295 cell_low[owner.nearest] = glm::min(cell_low[owner.nearest], band[b].voxel);
296 cell_high[owner.nearest] = glm::max(cell_high[owner.nearest], band[b].voxel);
297 }

299 // a thin pane never fits the cube minimum so a cell only needs to be as thick as its material
300 const float min_side = (2.0f * threshold) / voxel::k_voxel_size;
301 const auto min_cell_voxels = [&](size_t i) {
302 const glm::ivec3 extent = cell_high[i] - cell_low[i] + 1;
303 const float thickness = std::min(static_cast<float>(std::min({extent.x, extent.y, extent.z})), min_side);
304 return static_cast<uint32_t>(std::ceil(min_side * min_side * thickness));
305 };
306 const float release = std::clamp(values.shatter_release, 0.0f, 1.0f);
307 std::vector<bool> released(seeds.size(), false);
308 std::vector<size_t> eligible;
309 for (size_t i = 0; i < seeds.size(); ++i) {
310 if (cell_voxels[i] > 0 && cell_voxels[i] >= min_cell_voxels(i)) {
311 eligible.push_back(i);
312 released[i] = rng.chance(release);
313 }
314 }
315 if (eligible.empty()) {
316 out.stop = "cells too small to come off";
317 return out;
318 }
319 if (std::ranges::none_of(eligible, [&](size_t i) { return released[i]; })) {
320 released[eligible[rng.below(static_cast<uint32_t>(eligible.size()))]] = true;
321 }

327 const bool border = owner.gap < threshold && (mine || released[owner.second]);
328 bool crack = border;
329 crack = crack || (mine && touches_wall(band[b].voxel));
330 crack = crack ||
331 (mine && crater.pattern == FracturePattern::granular && hash01(crumble_seed, band[b].voxel) < values.shatter_crumble);

334 }

351inline auto planCoreSplit(std::span<const CoreVoxel> core, const Crater& crater) -> CoreSplit {
352 ZoneScopedN("physics::PlanCoreSplit");
353 CoreSplit out;
354 const FractureValues& values = crater.profile;
355 if (not(crater.fragment_energy > 0.0f) || not(crater.radius > 0.0f)) {
356 return out;
357 }
358 out.density = crater.fragment_energy / craterVolume(crater);
359 const float part_density = std::max(values.fragment_threshold * crater.toughness, 1.0e-9f);
360 if (out.density < part_density) {
361 return out;
362 }
363
364 Rng rng {streamSeed(crater.seed, FractureStream::core_split)};
365 const float doublings = std::log2(out.density / part_density);
366 const float parts_wanted =
367 std::clamp(2.0f + std::floor(doublings), 2.0f, static_cast<float>(std::max(values.fragment_max_parts, 2u)));
368
369 const float threshold = crackThickness();
370 const auto room = static_cast<uint32_t>(core.size() / minChunkVoxels(threshold));
371 const uint32_t parts = std::min(static_cast<uint32_t>(parts_wanted), room);
372 if (parts < 2) {
373 return out;
374 }
375
376 std::vector<glm::vec3> seeds {core[rng.below(static_cast<uint32_t>(core.size()))].center};
377 while (seeds.size() < parts) {
378 glm::vec3 best = seeds.front();
379 float best_gap = -1.0f;
380 for (int attempt = 0; attempt < 12; ++attempt) {
381 const glm::vec3 point = core[rng.below(static_cast<uint32_t>(core.size()))].center;
382 float gap = std::numeric_limits<float>::max();
383 for (const glm::vec3& s : seeds) {
384 gap = std::min(gap, glm::distance(s, point));
385 }
386 if (gap > best_gap) {
387 best_gap = gap;
388 best = point;
389 }
390 }
391 if (best_gap < 2.0f * threshold) {
392 break; // too small
393 }
394 seeds.push_back(best);
395 }
396 if (seeds.size() < 2) {
397 return out;
398 }
399 out.parts = static_cast<uint32_t>(seeds.size());
400
401 for (const CoreVoxel& c : core) {
402 float d1 = std::numeric_limits<float>::max();
403 float d2 = std::numeric_limits<float>::max();
404 for (const glm::vec3& seed_point : seeds) {
405 const float d = glm::distance(c.center, seed_point);
406 if (d < d1) {
407 d2 = d1;
408 d1 = d;
409 } else if (d < d2) {
410 d2 = d;
411 }
412 }
413 if (d2 - d1 < threshold) {
414 out.crack.push_back(c.voxel);
415 }
416 }
417 return out;
418}

422inline auto shardSeeds(const UnglueRegion& region, const Crater& crater) -> std::vector<glm::vec3> {
423 ZoneScopedN("physics::ShardSeeds");
424 std::vector<glm::vec3> seeds;
425 if (region.empty() || not patternShatters(crater.pattern)) {
426 return seeds;
427 }
428
429 // set order is not stable so the same hit could pick other seeds
430 std::vector<uint64_t> keys(region.keys.begin(), region.keys.end());
431 std::ranges::sort(keys);
432
433 const float gap = std::max(crater.profile.shatter_cell_size, crackThickness());
434 const float gap_squared = gap * gap;
435 const uint32_t max_cells = std::max(tunables().fracture_max_shatter_cells, 1u);
436 constexpr uint64_t k_axis_mask = (1ull << 21) - 1ull; // voxelKey packs 21 bits per axis
437 Rng rng {streamSeed(crater.seed, FractureStream::shard_points)};
438 for (uint32_t attempt = 0; attempt < max_cells * 12 && seeds.size() < max_cells; ++attempt) {
439 const uint64_t key = keys[rng.below(static_cast<uint32_t>(keys.size()))];
440 const glm::vec3 voxel {
441 static_cast<float>(key >> 42),
442 static_cast<float>((key >> 21) & k_axis_mask),
443 static_cast<float>(key & k_axis_mask),
444 };
445 const glm::vec3 point = (voxel + 0.5f) * voxel::k_voxel_size;
446 const bool crowded = std::ranges::any_of(seeds, [&](const glm::vec3& s) {
447 const glm::vec3 offset = s - point;
448 return glm::dot(offset, offset) < gap_squared;
449 });
450 if (not crowded) {
451 seeds.push_back(point);
452 }
453 }
454 return seeds;
455}
