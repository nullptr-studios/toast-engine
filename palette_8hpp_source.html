153inline auto validateLibrary(const MaterialLibrary& library) -> std::vector<TableProblem> {
154 std::vector<TableProblem> out;
155 if (library.materials.empty()) {
156 out.push_back({TableIssue::library_empty, 0});
157 return out;
158 }
159 if (library.materials.size() > k_max_physical_materials) {
160 out.push_back({TableIssue::library_too_large, static_cast<uint32_t>(library.materials.size())});
161 }
162
163 // !(x >= 0) also catches NaN
164 const auto negative = [](float value) { return !(value >= 0.0f); };
165
166 for (uint32_t i = 0; i < library.materials.size(); ++i) {
167 const PhysicalMaterial& m = library.materials[i];
168 if (m.density == 0) {
169 out.push_back({TableIssue::zero_density, i});
170 }
171 if (negative(m.toughness) || negative(m.structural_strength) || negative(m.shatter_radius) || negative(m.static_friction) ||
172 negative(m.dynamic_friction) || negative(m.restitution) || m.restitution > 1.0f || negative(m.ignition_energy) ||
173 negative(m.burn_rate) || negative(m.fuel)) {
174 out.push_back({TableIssue::out_of_range, i});
175 }
176 if (m.isIndestructible() && m.toughness == 0.0f) {
177 out.push_back({TableIssue::indestructible_with_zero_toughness, i});
178 }
179 }
180 return out;
181}

184inline auto validatePalette(const Palette& palette, const MaterialLibrary& library) -> std::vector<TableProblem> {
185 std::vector<TableProblem> out;
186 if (!(palette.entries[k_empty_palette_index] == PaletteEntry {})) {

196 }
197 }
