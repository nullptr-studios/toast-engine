138[[nodiscard]]
139inline auto resolve(const MassMoments& moments, float voxel_size = k_voxel_size) -> MassProperties {
140 MassProperties out;
141 if (moments.mass <= 0) {
142 return out;
143 }
144
145 const double s = voxel_size;
146 const double s3 = s * s * s;
147 const double s5 = s3 * s * s;
148
149 const double density_sum = static_cast<double>(moments.mass);
150 const double total_mass = density_sum * s3;
151
152 // sum(d * (l_a + 1/2) * (l_b + 1/2)) = m_ab + (m_a + m_b)/2 + density_sum/4
153 const auto product = [&](int64_t m_ab, int64_t m_a, int64_t m_b) {
154 return static_cast<double>(m_ab) + (0.5 * (static_cast<double>(m_a) + static_cast<double>(m_b))) + (0.25 * density_sum);
155 };
156
157 const double p_xx = product(moments.m_xx, moments.m_x, moments.m_x);
158 const double p_yy = product(moments.m_yy, moments.m_y, moments.m_y);
159 const double p_zz = product(moments.m_zz, moments.m_z, moments.m_z);
160 const double p_xy = product(moments.m_xy, moments.m_x, moments.m_y);
161 const double p_xz = product(moments.m_xz, moments.m_x, moments.m_z);
162 const double p_yz = product(moments.m_yz, moments.m_y, moments.m_z);
163
164 const double com_x = ((static_cast<double>(moments.m_x) / density_sum) + 0.5) * s;
165 const double com_y = ((static_cast<double>(moments.m_y) / density_sum) + 0.5) * s;
166 const double com_z = ((static_cast<double>(moments.m_z) / density_sum) + 0.5) * s;
167
168 // A cube of edge s adds m * s² / 6 to every diagonal
169 const double self_term = total_mass * s * s / 6.0;
170
171 double i_xx = (s5 * (p_yy + p_zz)) + self_term;
172 double i_yy = (s5 * (p_xx + p_zz)) + self_term;
173 double i_zz = (s5 * (p_xx + p_yy)) + self_term;
174 double i_xy = -s5 * p_xy;
175 double i_xz = -s5 * p_xz;
176 double i_yz = -s5 * p_yz;
177
178 i_xx -= total_mass * ((com_y * com_y) + (com_z * com_z));
179 i_yy -= total_mass * ((com_x * com_x) + (com_z * com_z));
180 i_zz -= total_mass * ((com_x * com_x) + (com_y * com_y));
181 i_xy += total_mass * com_x * com_y;
182 i_xz += total_mass * com_x * com_z;
183 i_yz += total_mass * com_y * com_z;
184
185 out.mass = static_cast<float>(total_mass);
186 out.center_of_mass = glm::vec3(static_cast<float>(com_x), static_cast<float>(com_y), static_cast<float>(com_z));
187 out.inertia = glm::mat3(
188 static_cast<float>(i_xx),
189 static_cast<float>(i_xy),
190 static_cast<float>(i_xz),
191 static_cast<float>(i_xy),
192 static_cast<float>(i_yy),
193 static_cast<float>(i_yz),
194 static_cast<float>(i_xz),
195 static_cast<float>(i_yz),
196 static_cast<float>(i_zz)
197 );
198 return out;
199}
