161constexpr auto maskMoments(const BrickOccupancy& mask) noexcept -> MassMoments {
162 // slice bit index is y * 8 + x, these pick the bits whose x or y has bit k set
163 constexpr std::array<uint64_t, 3> x_bit {0xAAAAAAAAAAAAAAAAull, 0xCCCCCCCCCCCCCCCCull, 0xF0F0F0F0F0F0F0F0ull};
164 constexpr std::array<uint64_t, 3> y_bit {0xFF00FF00FF00FF00ull, 0xFFFF0000FFFF0000ull, 0xFFFFFFFF00000000ull};
165
166 MassMoments out;
167 for (uint32_t z = 0; z < k_brick_dim; ++z) {
168 const uint64_t word = mask[z];
169 if (word == 0ull) {
170 continue;
171 }
172 const auto count = [word](uint64_t bits) { return static_cast<int64_t>(std::popcount(word & bits)); };
173 int64_t sx = 0;
174 int64_t sy = 0;
175 int64_t sxx = 0;
176 int64_t syy = 0;
177 int64_t sxy = 0;
178 for (uint32_t j = 0; j < 3; ++j) {
179 sx += count(x_bit[j]) << j;
180 sy += count(y_bit[j]) << j;
181 // a * b is the sum over bit pairs of 2^(j + k) when both bits are set
182 for (uint32_t k = 0; k < 3; ++k) {
183 sxx += count(x_bit[j] & x_bit[k]) << (j + k);
184 syy += count(y_bit[j] & y_bit[k]) << (j + k);
185 sxy += count(x_bit[j] & y_bit[k]) << (j + k);
186 }
187 }
188 const auto n = static_cast<int64_t>(std::popcount(word));
189 const auto lz = static_cast<int64_t>(z);
190 out.mass += n;
191 out.m_x += sx;
192 out.m_y += sy;
193 out.m_z += lz * n;
194 out.m_xx += sxx;
195 out.m_yy += syy;
196 out.m_zz += lz * lz * n;
197 out.m_xy += sxy;
198 out.m_xz += lz * sx;
199 out.m_yz += lz * sy;
200 }
201 return out;
202}

221[[nodiscard]]
222inline auto resolve(const MassMoments& moments, float voxel_size = k_voxel_size) -> MassProperties {
223 MassProperties out;
224 if (moments.mass <= 0) {
225 return out;
226 }
227
228 const double s = voxel_size;
229 const double s3 = s * s * s;
230 const double s5 = s3 * s * s;
231
232 const double density_sum = static_cast<double>(moments.mass);
233 const double total_mass = density_sum * s3;
234
235 // sum(d * (l_a + 1/2) * (l_b + 1/2)) = m_ab + (m_a + m_b)/2 + density_sum/4
236 const auto product = [&](int64_t m_ab, int64_t m_a, int64_t m_b) {
237 return static_cast<double>(m_ab) + (0.5 * (static_cast<double>(m_a) + static_cast<double>(m_b))) + (0.25 * density_sum);
238 };
239
240 const double p_xx = product(moments.m_xx, moments.m_x, moments.m_x);
241 const double p_yy = product(moments.m_yy, moments.m_y, moments.m_y);
242 const double p_zz = product(moments.m_zz, moments.m_z, moments.m_z);
243 const double p_xy = product(moments.m_xy, moments.m_x, moments.m_y);
244 const double p_xz = product(moments.m_xz, moments.m_x, moments.m_z);
245 const double p_yz = product(moments.m_yz, moments.m_y, moments.m_z);
246
247 const double com_x = ((static_cast<double>(moments.m_x) / density_sum) + 0.5) * s;
248 const double com_y = ((static_cast<double>(moments.m_y) / density_sum) + 0.5) * s;
249 const double com_z = ((static_cast<double>(moments.m_z) / density_sum) + 0.5) * s;
250
251 // A cube of edge s adds m * s² / 6 to every diagonal
252 const double self_term = total_mass * s * s / 6.0;
253
254 double i_xx = (s5 * (p_yy + p_zz)) + self_term;
255 double i_yy = (s5 * (p_xx + p_zz)) + self_term;
256 double i_zz = (s5 * (p_xx + p_yy)) + self_term;
257 double i_xy = -s5 * p_xy;
258 double i_xz = -s5 * p_xz;
259 double i_yz = -s5 * p_yz;
260
261 i_xx -= total_mass * ((com_y * com_y) + (com_z * com_z));
262 i_yy -= total_mass * ((com_x * com_x) + (com_z * com_z));
263 i_zz -= total_mass * ((com_x * com_x) + (com_y * com_y));
264 i_xy += total_mass * com_x * com_y;
265 i_xz += total_mass * com_x * com_z;
266 i_yz += total_mass * com_y * com_z;
267
268 out.mass = static_cast<float>(total_mass);
269 out.center_of_mass = glm::vec3(static_cast<float>(com_x), static_cast<float>(com_y), static_cast<float>(com_z));
270 out.inertia = glm::mat3(
271 static_cast<float>(i_xx),
272 static_cast<float>(i_xy),
273 static_cast<float>(i_xz),
274 static_cast<float>(i_xy),
275 static_cast<float>(i_yy),
276 static_cast<float>(i_yz),
277 static_cast<float>(i_xz),
278 static_cast<float>(i_yz),
279 static_cast<float>(i_zz)
280 );
281 return out;
282}
