141constexpr auto occupiedExtent(const BrickOccupancy& brick) noexcept -> BrickExtent {
142 uint64_t merged = 0;
143 uint32_t z_min = k_brick_dim;
144 uint32_t z_max = 0;
145 for (uint32_t z = 0; z < k_brick_dim; ++z) {
146 if (brick[z] != 0ull) {
147 merged |= brick[z];
148 z_min = z < z_min ? z : z_min;
149 z_max = z;
150 }
151 }
152
153 uint32_t columns = 0;
154 uint32_t y_min = k_brick_dim;
155 uint32_t y_max = 0;
156 for (uint32_t y = 0; y < k_brick_dim; ++y) {
157 const auto row = static_cast<uint32_t>((merged >> (y * k_brick_dim)) & 0xFFull);
158 if (row != 0u) {
159 columns |= row;
160 y_min = y < y_min ? y : y_min;
161 y_max = y;
162 }
163 }
164
165 return {
166 .min = { static_cast<uint32_t>(std::countr_zero(columns)), y_min, z_min},
167 .max = {31u - static_cast<uint32_t>(std::countl_zero(columns)), y_max, z_max},
168 };
169}
