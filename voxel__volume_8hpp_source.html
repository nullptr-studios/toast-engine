192 template<typename Visit>
193 void forEachSolidBrick(Visit&& visit) const {
194 for (size_t word_index = 0; word_index < m_solid_mask.size(); ++word_index) {
195 uint64_t word = m_solid_mask[word_index];
196 while (word != 0ull) {
197 const auto slot = static_cast<uint32_t>((word_index << 6u) + static_cast<size_t>(std::countr_zero(word)));
198 word &= word - 1ull;
199 visit(brickAtIndex(slot), slot);
200 }
201 }
