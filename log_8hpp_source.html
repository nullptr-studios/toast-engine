53constexpr auto splitFirstLine(std::string_view text) -> std::pair<std::string_view, std::string_view> {
54 const size_t newline = text.find('\n');
55 if (newline == std::string_view::npos) {
56 return {text, {}};
57 }
58 std::string_view first = text.substr(0, newline);
59 if (!first.empty() && first.back() == '\r') {
60 first.remove_suffix(1);
61 }
62 return {first, text.substr(newline + 1)};
63}
