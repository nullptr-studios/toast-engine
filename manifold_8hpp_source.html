46[[nodiscard]]
47inline auto operator<(const Manifold& lhs, const Manifold& rhs) -> bool {
48 if (lhs.pair != rhs.pair) {
49 return lhs.pair < rhs.pair;
50 }
51
52 if (lhs.normal_index != rhs.normal_index) {
53 return lhs.normal_index < rhs.normal_index;
54 }
55
56 if (lhs.contact_count != rhs.contact_count) {
57 return lhs.contact_count < rhs.contact_count;
58 }
59
60 for (std::size_t index = 0; index < lhs.contact_count && index < lhs.contacts.size(); ++index) {
61 const ContactPoint& lhs_contact = lhs.contacts[index];
62 const ContactPoint& rhs_contact = rhs.contacts[index];
63
64 if (lhs_contact.feature_a != rhs_contact.feature_a) {
65 return lhs_contact.feature_a < rhs_contact.feature_a;
66 }
67 if (lhs_contact.feature_b != rhs_contact.feature_b) {
68 return lhs_contact.feature_b < rhs_contact.feature_b;
69 }
70 }
71
72 return false;
73}
