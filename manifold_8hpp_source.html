49[[nodiscard]]
50inline auto operator<(const Manifold& lhs, const Manifold& rhs) -> bool {
51 if (lhs.pair != rhs.pair) {
52 return lhs.pair < rhs.pair;
53 }
54
55 if (lhs.normal_index != rhs.normal_index) {
56 return lhs.normal_index < rhs.normal_index;
57 }
58
59 if (lhs.contact_count != rhs.contact_count) {
60 return lhs.contact_count < rhs.contact_count;
61 }
62
63 for (std::size_t index = 0; index < lhs.contact_count && index < lhs.contacts.size(); ++index) {
64 const ContactPoint& lhs_contact = lhs.contacts[index];
65 const ContactPoint& rhs_contact = rhs.contacts[index];
66
67 if (lhs_contact.feature_a != rhs_contact.feature_a) {
68 return lhs_contact.feature_a < rhs_contact.feature_a;
69 }
70 if (lhs_contact.feature_b != rhs_contact.feature_b) {
71 return lhs_contact.feature_b < rhs_contact.feature_b;
72 }
73 }
74
75 return false;
76}
