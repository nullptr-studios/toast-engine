368 template<typename R = void, typename... Args>
369 auto call(std::string_view name, Args&&... args) -> R {
370 std::array<std::any, sizeof...(Args)> any_args {std::any(std::decay_t<Args>(args))...};
371 if constexpr (std::is_void_v<R>) {
372 if (m_info && m_info->getMethod(name)) {
373 _detail::callMethodChain(m_info, this, name, args...);
374 }
375 _detail::callNodeScripts(this, name, any_args);
376 } else {
377 R result {};
378 if (m_info && m_info->getMethod(name)) {
379 result = _detail::callMethodChain<R>(m_info, this, name, args...);
380 }
381 _detail::callNodeScripts(this, name, any_args);
382 return result;
383 }
384 }
