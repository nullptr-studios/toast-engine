320 auto call(std::string_view name, Args&&... args) -> R {
321 std::array<std::any, sizeof...(Args)> any_args {std::any(std::decay_t<Args>(args))...};
322 if constexpr (std::is_void_v<R>) {
323 if (m_info && m_info->getMethod(name)) {
324 _detail::callMethodChain(m_info, this, name, args...);
