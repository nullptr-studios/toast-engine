30 [[nodiscard]]
31 constexpr auto isMirror() const noexcept -> bool {
32 // Permutation parity times one sign flip per negated axis
33 int inversions = 0;
34 for (int i = 0; i < 3; ++i) {
35 for (int j = i + 1; j < 3; ++j) {
36 inversions += source[i] > source[j] ? 1 : 0;
37 }
38 }
39 const int negated = static_cast<int>(flip[0]) + static_cast<int>(flip[1]) + static_cast<int>(flip[2]);
40 return (inversions + negated) % 2 != 0;
41 }
42};

45inline auto allLatticeOrientations() -> std::array<LatticeOrientation, 48> {
46 constexpr std::array<std::array<uint8_t, 3>, 6> permutations {
47 {{0, 1, 2}, {0, 2, 1}, {1, 0, 2}, {1, 2, 0}, {2, 0, 1}, {2, 1, 0}}
48 };
49
50 std::array<LatticeOrientation, 48> out {};
51 size_t next = 0;
52 for (const std::array<uint8_t, 3>& permutation : permutations) {
53 for (uint32_t signs = 0; signs < 8; ++signs) {
54 LatticeOrientation orientation;
55 orientation.source = permutation;
56 orientation.flip = {(signs & 1u) != 0, (signs & 2u) != 0, (signs & 4u) != 0};
57 out[next++] = orientation;
58 }
59 }
60 return out;

94inline auto placementFromTransform(const glm::mat4& transform, float voxel_size = k_voxel_size, float tolerance = 1e-4f)
95 -> std::optional<LatticePlacement> {
96 LatticePlacement out;
97 std::array<bool, 3> used {false, false, false};
98
99 for (int row = 0; row < 3; ++row) {
100 int found = -1;
101 for (int column = 0; column < 3; ++column) {
102 const float value = transform[column][row];
103 if (std::abs(std::abs(value) - 1.0f) <= tolerance) {
104 if (found != -1) {
105 return std::nullopt;
106 }
107 found = column;
108 out.orientation.flip[row] = value < 0.0f;
109 } else if (std::abs(value) > tolerance) {
110 return std::nullopt;
111 }
112 }
113 if (found == -1 || used[found]) {
114 return std::nullopt;
115 }
116 used[found] = true;
117 out.orientation.source[row] = static_cast<uint8_t>(found);
118 }
119
120 if (std::abs(transform[0][3]) > tolerance || std::abs(transform[1][3]) > tolerance || std::abs(transform[2][3]) > tolerance ||
121 std::abs(transform[3][3] - 1.0f) > tolerance) {
122 return std::nullopt;
123 }
124
125 const float slack = 0.001f / voxel_size;
126 for (int axis = 0; axis < 3; ++axis) {
127 const float voxels = transform[3][axis] / voxel_size;
128 const float nearest = std::round(voxels);
129 if (std::abs(voxels - nearest) > slack) {
130 return std::nullopt;
131 }
132 out.offset[axis] = static_cast<int32_t>(nearest);
133 }
134 return out;
135}

157 PaletteRemap out;
158 for (uint32_t i = 1; i < k_palette_size; ++i) {
159 const PaletteEntry& entry = piece.entries[i];
160 if (entry == PaletteEntry {}) {
161 continue;
162 }
163
164 bool matched = false;
165 if (entry.emissive == 0 || piece.max_emissive == master.max_emissive) {
166 for (uint32_t j = 1; j < k_palette_size; ++j) {
167 if (master.entries[j] == entry) {
168 out.table[i] = static_cast<uint8_t>(j);
169 matched = true;
170 break;
171 }
172 }
173 }
174 if (!matched) {
175 out.unmatched.push_back(static_cast<uint8_t>(i));
176 }
177 }
