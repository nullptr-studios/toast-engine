150template<typename T>
151auto readEventIndex(lua_State* state, int index, std::string_view type, T& out) -> bool {
152 if constexpr (!std::same_as<T, bool> && (std::is_enum_v<T> || std::integral<T>)) {
153 int64_t value = 0;
154 if (!readEventPrimitive(state, index, value)) {
155 return false;
156 }
157 out = static_cast<T>(value);
158 return true;
159 } else if constexpr (std::floating_point<T>) {
160 double value = 0.0;
161 if (!readEventPrimitive(state, index, value)) {
162 return false;
163 }
164 out = static_cast<T>(value);
165 return true;
166 } else if constexpr (is_vector_v<T>) {
167 const int64_t count = eventArrayLength(state, index);
168 if (count < 0) {
169 return false;
170 }
171 out.clear();
172 out.reserve(static_cast<size_t>(count));
173 for (int64_t i = 1; i <= count; ++i) {
174 typename T::value_type element {};
175 const bool ok = readEventIndex(state, pushEventElement(state, index, i), type, element);
176 popEventValue(state);
177 if (!ok) {
178 return false;
179 }
180 out.push_back(std::move(element));
181 }
182 return true;
183 } else if constexpr (is_handle_v<T>) {
184 assets::HandleBase handle;
185 if (!readEventHandle(state, index, type, handle)) {
186 return false;
187 }
188 out = T(handle.hasValue() ? handle.operator->() : nullptr, handle.uid(), handle.path());
189 return true;
190 } else {
191 return readEventPrimitive(state, index, out);
192 }
193}

196auto readEventValue(lua_State* state, int table, std::string_view field, std::string_view type, T& out, std::string& error)
197 -> bool {
198 const bool ok = readEventIndex(state, pushEventField(state, table, field), type, out);
199 popEventValue(state);
200 if (!ok) {
201 error = "invalid or missing field '" + std::string(field) + "'";
202 }

206template<typename T>
207void pushEventValue(lua_State* state, const T& value) {
208 if constexpr (!std::same_as<T, bool> && (std::is_enum_v<T> || std::integral<T>)) {
209 pushEventPrimitive(state, static_cast<int64_t>(value));
210 } else if constexpr (std::floating_point<T>) {
211 pushEventPrimitive(state, static_cast<double>(value));
212 } else if constexpr (std::same_as<T, std::string>) {
213 pushEventPrimitive(state, std::string_view(value));
214 } else if constexpr (is_vector_v<T>) {
215 pushEventArray(state, value.size());
216 int64_t i = 0;
217 for (const auto& element : value) {
218 pushEventValue<typename T::value_type>(state, element);
219 setEventArrayElement(state, ++i);
220 }
221 } else if constexpr (is_handle_v<T>) {
222 pushEventPrimitive(state, static_cast<const assets::HandleBase&>(value));
223 } else {
224 pushEventPrimitive(state, value);
225 }
226}

228template<typename Event, typename... Fields, size_t... I>
229auto sendEvent(lua_State* state, int payload, std::string& error, const std::tuple<Fields...>& fields, std::index_sequence<I...>)
230 -> bool {
231 std::tuple<typename Fields::value_type...> values;
232 const bool valid =
233 (readEventValue(state, payload, std::get<I>(fields).name, std::get<I>(fields).type, std::get<I>(values), error) && ...);
234 if (!valid) {
235 return false;
236 }
237 if constexpr (std::is_constructible_v<Event, decltype(std::move(std::get<I>(values)))...>) {
238 std::apply([](auto&&... args) { event::send<Event>(std::move(args)...); }, values);
239 } else {
240 Event built {};
241 (std::get<I>(fields).set(built, std::move(std::get<I>(values))), ...);
242 event::send<Event>(std::move(built));
243 }
244 return true;
245}

258template<typename Event, bool Sendable = true, typename... Fields>
259void registerLuaEvent(std::string_view name, Fields... fields) {
260 if constexpr (Sendable) {
261 static_assert(
262 std::is_default_constructible_v<Event> || std::is_constructible_v<Event, typename Fields::value_type...>,
263 "Lua events need a constructor taking the public fields in order, or a default constructor"
264 );
265 }
266 auto metadata = std::make_shared<std::tuple<Fields...>>(fields...);
267 auto binding = std::make_shared<LuaEventBinding>();
268 binding->name = name;
269 if constexpr (Sendable) {
270 binding->send = [metadata](lua_State* state, int payload, std::string& error) {
271 return _detail::sendEvent<Event>(state, payload, error, *metadata, std::index_sequence_for<Fields...> {});
272 };
273 }
274 binding->subscribe = [metadata](event::Listener& listener, std::string subscription_name, LuaEventBinding::Callback callback) {
275 listener.subscribe<Event>(std::move(subscription_name), [metadata, callback = std::move(callback)](Event& value) {
276 return callback([metadata, &value](lua_State* state) {
277 _detail::pushEvent(state, value, *metadata, std::index_sequence_for<Fields...> {});
278 });
279 });
280 };
281 binding->unsubscribe = [](event::Listener& listener, std::string_view subscription_name) {
282 listener.unsubscribe<Event>(subscription_name);
283 };
284 LuaEventRegistry::registerBinding(std::move(binding));
285}
