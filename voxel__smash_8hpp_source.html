28inline auto segmentBoxDistanceSquared(const glm::vec3& from, const glm::vec3& to, const glm::vec3& center, float half_extent)
29 -> float {
30 const auto squared = [&](float t) {
31 const glm::vec3 gap = glm::max(glm::abs(glm::mix(from, to, t) - center) - glm::vec3(half_extent), glm::vec3(0.0f));
32 return glm::dot(gap, gap);
33 };
34
35 float low = 0.0f;
36 float high = 1.0f;
37 for (int i = 0; i < 32; ++i) {
38 const float first = low + ((high - low) / 3.0f);
39 const float second = high - ((high - low) / 3.0f);
40 if (squared(first) < squared(second)) {
41 high = second;
42 } else {
43 low = first;
44 }
45 }
46 return squared(0.5f * (low + high));
47}

49struct SmashVolume {
50 glm::vec3 axis_start {};
51 glm::vec3 axis_end {};
52 glm::vec3 sweep {};
53 float reach = 0.0f;
54 glm::vec3 up {0.0f, 0.0f, 1.0f};
55 float floor = k_no_floor;
56 bool touching = false;
57
58 [[nodiscard]]
59 auto poseCount() const -> int {
60 if (touching || not(reach > 0.0f)) {
61 return 1;
62 }
63 return std::max(1, static_cast<int>(std::ceil(glm::length(sweep) / (0.25f * reach))));
64 }
65
66 [[nodiscard]]
67 auto contains(const glm::vec3& point, int poses) const -> bool {
68 if (glm::dot(point, up) < floor) {
69 return false;
70 }
71 if (touching) {
72 return segmentBoxDistanceSquared(axis_start, axis_start + sweep, point, 0.5f * voxel::k_voxel_size) <=
73 (reach * reach) + 1.0e-8f;
74 }
75
76 const glm::vec3 axis = axis_end - axis_start;
77 const float axis_squared = glm::dot(axis, axis);
78 const float reach_squared = reach * reach;
79 for (int pose = 0; pose <= poses; ++pose) {
80 const glm::vec3 offset = point - (axis_start + (sweep * (static_cast<float>(pose) / static_cast<float>(poses))));
81 const float along = axis_squared > 0.0f ? glm::clamp(glm::dot(offset, axis) / axis_squared, 0.0f, 1.0f) : 0.0f;
82 const glm::vec3 from_axis = offset - (axis * along);
83 if (glm::dot(from_axis, from_axis) <= reach_squared) {
84 return true;
85 }
86 }
87 return false;
88 }
89
90 [[nodiscard]]
91 auto bounds() const -> std::pair<glm::vec3, glm::vec3> {
92 const glm::vec3 low = glm::min(axis_start, axis_end);
93 const glm::vec3 high = glm::max(axis_start, axis_end);
94 return {glm::min(low, low + sweep) - glm::vec3(reach), glm::max(high, high + sweep) + glm::vec3(reach)};
95 }
96};

105inline auto collectSmashPieces(

110 if (materials.materials.empty()) {
111 return out;
112 }
113
114 const auto [low, high] = smash.bounds();
115 const glm::ivec3 voxel_dims = glm::ivec3(volume.voxelDims());
116 glm::ivec3 first = glm::ivec3(glm::floor(low / voxel::k_voxel_size));
117 glm::ivec3 last = glm::ivec3(glm::ceil(high / voxel::k_voxel_size)) - 1;
118 first = glm::clamp(first, glm::ivec3(0), voxel_dims - 1);
119 last = glm::clamp(last, glm::ivec3(0), voxel_dims - 1);
120 if (first.x > last.x || first.y > last.y || first.z > last.z) {
121 return out;
122 }
123
124 const auto brick_dim = static_cast<int32_t>(voxel::k_brick_dim);
125 const glm::ivec3 first_brick = first / brick_dim;
126 const glm::ivec3 last_brick = last / brick_dim;
127 const int poses = smash.poseCount();
128
129 for (int32_t bz = first_brick.z; bz <= last_brick.z; ++bz) {
130 for (int32_t by = first_brick.y; by <= last_brick.y; ++by) {
131 for (int32_t bx = first_brick.x; bx <= last_brick.x; ++bx) {
132 const glm::ivec3 brick {bx, by, bz};
133 const voxel::BrickOccupancy* occupancy = volume.occupancyPointer(brick);
134 if (occupancy == nullptr) {
135 continue;
136 }
137
138 voxel::BrickOccupancy mask {};
139 const glm::ivec3 brick_first = glm::max(first, brick * brick_dim);
140 const glm::ivec3 brick_last = glm::min(last, (brick * brick_dim) + (brick_dim - 1));
141 for (int32_t z = brick_first.z; z <= brick_last.z; ++z) {
142 for (int32_t y = brick_first.y; y <= brick_last.y; ++y) {
143 for (int32_t x = brick_first.x; x <= brick_last.x; ++x) {
144 const glm::ivec3 voxel {x, y, z};
145 const glm::ivec3 local = voxel - (brick * brick_dim);
146 if (not voxel::isSolid(
147 *occupancy, static_cast<uint32_t>(local.x), static_cast<uint32_t>(local.y), static_cast<uint32_t>(local.z)
148 )) {
149 continue;
150 }
151
152 const glm::vec3 center = (glm::vec3 {voxel} + 0.5f) * voxel::k_voxel_size;
153 if (not smash.contains(center, poses)) {
154 continue;
155 }
156
157 const uint8_t palette_index = volume.materialAt(voxel);
158 const voxel::PhysicalMaterial& material =
159 materials.materials[voxel::resolveMaterialIndex(palette, materials, palette_index)];
160 if (material.isIndestructible() || material.toughness > energy) {
161 continue;
162 }
163
164 voxel::setSolid(
165 mask, static_cast<uint32_t>(local.x), static_cast<uint32_t>(local.y), static_cast<uint32_t>(local.z), true
166 );
167 out.index_sum += glm::vec3(voxel);
168 ++out.voxels;
169 }
170 }
171 }
172
173 if (voxel::isEmpty(mask)) {
174 continue;
175 }
176 for (const voxel::BrickOccupancy& region : voxel::brickComponents(mask)) {
177 DetachedComponent piece;
178 piece.component = static_cast<uint32_t>(out.pieces.size());
179 piece.voxel_count = voxel::popCount(region);
180 piece.pieces.push_back(voxel::BrickPiece {.brick = brick, .voxels = region, .component = piece.component});
181 out.pieces.push_back(std::move(piece));
182 }
183 }
184 }
185 }
186
187 return out;
188}
