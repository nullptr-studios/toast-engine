291 template<typename R = void, typename... Args>
292 auto call(void* obj, std::string_view method_name, Args&&... args) const -> R {
293 if (!getMethod(method_name)) {
294 TOAST_WARN("Reflect", "call(): reflected function '{}' not found on '{}'", method_name, type);
295 assert(false && "NodeInfo::call(): reflected function not found");
296 if constexpr (std::is_void_v<R>) {
297 return;
298 } else {
299 return R {};
300 }
301 }
302 // Build an any-args array so args are forwarded to Lua scripts as well
303 std::array<std::any, sizeof...(Args)> any_args {std::any(std::decay_t<Args>(args))...};
304 if constexpr (std::is_void_v<R>) {
305 _detail::callMethodChain(this, obj, method_name, std::forward<Args>(args)...);
306 _detail::callNodeScripts(static_cast<Node*>(obj), method_name, any_args);
307 } else {
308 R result = _detail::callMethodChain<R>(this, obj, method_name, std::forward<Args>(args)...);
309 _detail::callNodeScripts(static_cast<Node*>(obj), method_name, any_args);
310 return result;
311 }
312 }

433// Walks base->derived, calling the method at every level where it is defined
434template<typename R, typename... Args>
435auto callMethodChain(const NodeInfo* info, void* obj, std::string_view method_name, const Args&... args) -> R {
436 if constexpr (std::is_void_v<R>) {
437 if (info->base_type) {
438 callMethodChain(info->base_type, obj, method_name, args...);
439 }
440 for (const auto& m : info->methods) {
441 if (m.name == method_name) {
442 m.call(obj, args...);
443 break;
444 }
445 }
446 } else {
447 R result {};
448 if (info->base_type) {
449 result = callMethodChain<R>(info->base_type, obj, method_name, args...);
450 }
451 for (const auto& m : info->methods) {
452 if (m.name == method_name) {
453 result = m.call<R>(obj, args...);
454 break;
455 }
456 }
457 return result;
458 }
459}
