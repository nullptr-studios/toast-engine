289 template<typename R = void, typename... Args>
290 auto call(void* obj, std::string_view method_name, Args&&... args) const -> R {
291 if (!getMethod(method_name)) {
292 TOAST_WARN("Reflect", "call(): reflected function '{}' not found on '{}'", method_name, type);
293 assert(false && "NodeInfo::call(): reflected function not found");
294 if constexpr (std::is_void_v<R>) {
295 return;
296 } else {
297 return R {};
298 }
299 }
300 // Build an any-args array so args are forwarded to Lua scripts as well
301 std::array<std::any, sizeof...(Args)> any_args {std::any(std::decay_t<Args>(args))...};
302 if constexpr (std::is_void_v<R>) {
303 _detail::callMethodChain(this, obj, method_name, std::forward<Args>(args)...);
304 _detail::callNodeScripts(static_cast<Node*>(obj), method_name, any_args);
305 } else {
306 R result = _detail::callMethodChain<R>(this, obj, method_name, std::forward<Args>(args)...);
307 _detail::callNodeScripts(static_cast<Node*>(obj), method_name, any_args);
308 return result;
309 }
310 }

425// Walks base->derived, calling the method at every level where it is defined
426template<typename R, typename... Args>
427auto callMethodChain(const NodeInfo* info, void* obj, std::string_view method_name, const Args&... args) -> R {
428 if constexpr (std::is_void_v<R>) {
429 if (info->base_type) {
430 callMethodChain(info->base_type, obj, method_name, args...);
431 }
432 for (const auto& m : info->methods) {
433 if (m.name == method_name) {
434 m.call(obj, args...);
435 break;
436 }
437 }
438 } else {
439 R result {};
440 if (info->base_type) {
441 result = callMethodChain<R>(info->base_type, obj, method_name, args...);
442 }
443 for (const auto& m : info->methods) {
444 if (m.name == method_name) {
445 result = m.call<R>(obj, args...);
446 break;
447 }
448 }
449 return result;
450 }
451}
