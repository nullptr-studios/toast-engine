209template<typename T>
210class Setting {
211public:
212 Setting() = default;
213
214 explicit Setting(Entry* entry) : m_entry(entry) { }
215
216 [[nodiscard]]
217 auto valid() const -> bool {
218 return m_entry != nullptr;
219 }
220
221 [[nodiscard]]
222 auto value() const -> T {
223 if (m_entry == nullptr) {
224 return T {};
225 }
226 // Reads the entry directly. Going back through value(key) hashed the key and built a std::string to
227 // look up the entry this handle is already holding
228 const Value resolved = Settings::get().resolved(m_entry);
229 if (const auto* typed = std::get_if<T>(&resolved)) {
230 return *typed;
231 }
232 return T {};
233 }
234
235 operator T() const { // NOLINT(google-explicit-constructor) - the whole point is reading it like a T
236 return value();
237 }
238
239 void set(T v) {
240 if (m_entry != nullptr) {
241 Settings::get().setValue(m_entry->key, Value {std::move(v)});
242 }
243 }
244
249 void onChange(std::function<void(T)> callback) {
250 if (m_entry == nullptr) {
251 return;
252 }
253 auto wrapper = [callback = std::move(callback)](const Value& v) {
254 if (const auto* typed = std::get_if<T>(&v)) {
255 callback(*typed);
256 }
257 };
258 wrapper(Value {value()});
259 m_entry->on_change.emplace_back(std::move(wrapper));
260 }
261
262 [[nodiscard]]
263 auto entry() const -> Entry* {
264 return m_entry;
265 }
266
267private:
268 Entry* m_entry = nullptr;
269};
