65[[nodiscard]]
66inline auto clusterDebris(std::span<const DebrisVoxel> voxels, const glm::ivec3& jitter, Rng& rng, float split_chance)
67 -> std::vector<DebrisCluster> {
68 struct Entry {
69 glm::ivec3 cell;
70 uint32_t bit;
71 uint8_t material;
72 };
73
74 std::vector<Entry> entries;
75 entries.reserve(voxels.size());
76 for (const DebrisVoxel& v : voxels) {
77 const glm::ivec3 shifted = v.voxel - jitter;
78 const glm::ivec3 cell {shifted.x >> 1, shifted.y >> 1, shifted.z >> 1};
79 entries.push_back({.cell = cell, .bit = debrisBit(shifted - (cell * 2)), .material = v.material});
80 }
81 std::ranges::sort(entries, [](const Entry& a, const Entry& b) {
82 if (a.cell.z != b.cell.z) {
83 return a.cell.z < b.cell.z;
84 }
85 if (a.cell.y != b.cell.y) {
86 return a.cell.y < b.cell.y;
87 }
88 if (a.cell.x != b.cell.x) {
89 return a.cell.x < b.cell.x;
90 }
91 return a.bit < b.bit;
92 });
93
94 std::vector<DebrisCluster> out;
95 for (size_t first = 0; first < entries.size();) {
96 size_t last = first;
97 std::array<uint8_t, 8> material {};
98 uint32_t present = 0;
99 while (last < entries.size() && entries[last].cell == entries[first].cell) {
100 present |= 1u << entries[last].bit;
101 material[entries[last].bit] = entries[last].material;
102 ++last;
103 }
104
105 // a block sometimes breaks into 2 or 3 smaller pieces
106 const auto count = static_cast<uint32_t>(std::popcount(present));
107 uint32_t parts = 1;
108 if (count > 1 && rng.chance(split_chance)) {
109 parts = (count > 2 && rng.chance(0.5f)) ? 3u : 2u;
110 }
111 std::array<uint32_t, 3> part_masks {};
112 for (uint32_t bit = 0; bit < 8; ++bit) {
113 if ((present & (1u << bit)) != 0) {
114 part_masks[parts == 1 ? 0 : rng.below(parts)] |= 1u << bit;
115 }
116 }
117
118 for (uint32_t p = 0; p < parts; ++p) {
119 if (part_masks[p] == 0) {
120 continue;
121 }
122 // most common material of the part
123 uint8_t best = 0;
124 int best_count = 0;
125 for (uint32_t bit = 0; bit < 8; ++bit) {
126 if ((part_masks[p] & (1u << bit)) == 0) {
127 continue;
128 }
129 int same = 0;
130 for (uint32_t other = 0; other < 8; ++other) {
131 same += ((part_masks[p] & (1u << other)) != 0 && material[other] == material[bit]) ? 1 : 0;
132 }
133 if (same > best_count) {
134 best_count = same;
135 best = material[bit];
136 }
137 }
138 out.push_back(
139 DebrisCluster {
140 .origin = (entries[first].cell * 2) + jitter,
141 .mask = static_cast<uint8_t>(part_masks[p]),
142 .material = best,
143 }
144 );
145 }
146 first = last;
147 }
148 return out;
149}

151[[nodiscard]]
152inline auto debrisSpeed(float energy, float mass, float scale) -> float {
153 if (not(energy > 0.0f) || not(mass > 0.0f) || not(scale > 0.0f)) {
154 return 0.0f;
155 }
156 return std::min(std::sqrt(2.0f * energy / mass) * scale, k_max_debris_speed);
157}

160inline auto debrisConeDirection(const glm::vec3& axis, Rng& rng, float spread) -> glm::vec3 {
161 const glm::vec3 tilted = axis + (rng.direction() * spread);
162 const float length = glm::length(tilted);
163 return length > 1.0e-4f ? tilted / length : axis;
