34constexpr auto canWrite(uint8_t current, WriteMode mode, uint8_t match_id) noexcept -> bool {
35 switch (mode) {
36 case WriteMode::replace: return true;
37 case WriteMode::empty_only: return current == k_empty_palette_index;
38 case WriteMode::solid_only: return current != k_empty_palette_index;
39 case WriteMode::match: return current == match_id;
40 }
41 return false;
42}

45constexpr auto canAdd(uint8_t id, WriteMode mode, uint8_t match_id) noexcept -> bool {
46 return id != k_empty_palette_index && canWrite(k_empty_palette_index, mode, match_id);
