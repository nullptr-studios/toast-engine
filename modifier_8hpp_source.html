62class DeadzoneModifier : public IModifier {
63public:
64 explicit DeadzoneModifier(const toml::table& t)
65 : m_lower(parse::number(t["lower_threshold"], 0.2f)),
66 m_upper(parse::number(t["upper_threshold"], 1.0f)) { }
67
68 void apply(Value& value, const EvalContext& /*ctx*/) override {
69 value.data.x = remap(value.data.x);
70 value.data.y = remap(value.data.y);
71 }
72
73 [[nodiscard]]
74 auto name() const -> std::string_view override {
75 return "deadzone";
76 }
77
78private:
79 [[nodiscard]]
80 auto remap(float v) const -> float {
81 const float mag = std::abs(v);
82 if (mag < m_lower) {
83 return 0.0f;
84 }
85 const float span = std::max(m_upper - m_lower, 0.0001f);
86 const float t = glm::clamp((mag - m_lower) / span, 0.0f, 1.0f);
87 return std::copysign(t, v);
88 }
89
90 float m_lower = 0.2f;
91 float m_upper = 1.0f;
92};

204 m_y(parse::boolean(t["y"], true)),

206
207 void apply(Value& value, const EvalContext& /*ctx*/) override {
208 float delta = Time::delta();
209 const float t = glm::clamp(m_speed * delta, 0.0f, 1.0f);

212 value.data.x = m_current.x;
213 }
214 if (m_y) {

216 value.data.y = m_current.y;
217 }
218 }

222 return "smooth";
223 }

244 (!m_shift || detail::modifierHeld(ctx.mods, ModifierKey::shift)) &&
