35inline auto normalIndexOf(glm::ivec3 direction) noexcept -> uint8_t {
36 if (direction.x == 0 && direction.y == 0 && direction.z == 0) {
37 return k_normal_undefined;
38 }
39 const int raw = (direction.x + 1) + ((direction.y + 1) * 3) + ((direction.z + 1) * 9);

69[[nodiscard]]
70inline auto classifyFromNeighbours(
71 bool solid_neg_x, bool solid_pos_x, bool solid_neg_y, bool solid_pos_y, bool solid_neg_z, bool solid_pos_z
72) noexcept -> uint8_t {
73 const int exposed = static_cast<int>(!solid_neg_x) + static_cast<int>(!solid_pos_x) + static_cast<int>(!solid_neg_y) +
74 static_cast<int>(!solid_pos_y) + static_cast<int>(!solid_neg_z) + static_cast<int>(!solid_pos_z);
75 if (exposed == 0) {
76 return packClassification(VoxelClass::inside, k_normal_undefined);
77 }
78
79 // subtracting flags cancels to (0,0,0) for an isolated voxel which reads as undefined and gets dropped
80 const auto axis = [](bool solid_pos, bool solid_neg) -> int {
81 if (!solid_pos) {
82 return 1;
83 }
84 return solid_neg ? 0 : -1;
85 };
86 const glm::ivec3 direction(axis(solid_pos_x, solid_neg_x), axis(solid_pos_y, solid_neg_y), axis(solid_pos_z, solid_neg_z));
87 const uint8_t normal = normalIndexOf(direction);
88
89 VoxelClass type = VoxelClass::corner;
90 if (normal != k_normal_undefined) {
91 if (exposed == 1) {
92 type = VoxelClass::face;
93 } else if (exposed == 2) {
94 type = VoxelClass::edge;
95 }
96 }
97 return packClassification(type, normal);
98}

131inline void
132 buildBrickSurfaceInto(const BrickOccupancy& brick, const BrickNeighbourhood& neighbours, std::vector<SurfaceVoxel>& out) {
133 const BrickOccupancy neg_x = neighboursNegX(brick, neighbours.neg_x);
134 const BrickOccupancy pos_x = neighboursPosX(brick, neighbours.pos_x);
135 const BrickOccupancy neg_y = neighboursNegY(brick, neighbours.neg_y);
136 const BrickOccupancy pos_y = neighboursPosY(brick, neighbours.pos_y);
137 const BrickOccupancy neg_z = neighboursNegZ(brick, neighbours.neg_z);
138 const BrickOccupancy pos_z = neighboursPosZ(brick, neighbours.pos_z);
139
140 const BrickOccupancy shell = brick & ~(neg_x & pos_x & neg_y & pos_y & neg_z & pos_z);
141
142 for (uint32_t z = 0; z < k_brick_dim; ++z) {
143 uint64_t word = shell[z];
144 while (word != 0ull) {
145 const auto bit = static_cast<uint32_t>(std::countr_zero(word));
146 word &= word - 1ull;
147 const uint64_t mask = 1ull << bit;
148
149 const bool solid_neg_x = (neg_x[z] & mask) != 0ull;
150 const bool solid_pos_x = (pos_x[z] & mask) != 0ull;
151 const bool solid_neg_y = (neg_y[z] & mask) != 0ull;
152 const bool solid_pos_y = (pos_y[z] & mask) != 0ull;
153 const bool solid_neg_z = (neg_z[z] & mask) != 0ull;
154 const bool solid_pos_z = (pos_z[z] & mask) != 0ull;
155
156 SurfaceVoxel entry;
157 entry.local_index = static_cast<uint16_t>((z * 64u) + bit);
158 entry.classification = classifyFromNeighbours(solid_neg_x, solid_pos_x, solid_neg_y, solid_pos_y, solid_neg_z, solid_pos_z);
159 entry.exposure = exposureMask(solid_neg_x, solid_pos_x, solid_neg_y, solid_pos_y, solid_neg_z, solid_pos_z);
160 out.push_back(entry);
161 }
162 }
163}
