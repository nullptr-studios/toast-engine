202 template<typename R = void, typename... Args>
203 auto call(void* obj, Args&&... args) const -> R {
204 bool signature_ok =
205 parameters.size() == sizeof...(Args) && return_type_id != nullptr && *return_type_id == typeid(std::decay_t<R>);
206 if (signature_ok) {
207 const std::array<const std::type_info*, sizeof...(Args)> expected = {&typeid(std::decay_t<Args>)...};
208 for (std::size_t i = 0; i < expected.size(); ++i) {
209 if (parameters[i].type_id == nullptr || *parameters[i].type_id != *expected[i]) {
210 signature_ok = false;
211 break;
212 }
213 }
214 }
215 if (!signature_ok) {
216 TOAST_WARN("Reflect", "call(): signature mismatch for '{}'; invoking is undefined behavior", name);
217 assert(false && "FunctionInfo::call(): argument or return type mismatch");
218 }
219
220 using Trampoline = R (*)(void*, std::decay_t<Args>...);
221 auto* trampoline = reinterpret_cast<Trampoline>(invoke);
222 return trampoline(obj, std::forward<Args>(args)...);
223 }
224
233 [[nodiscard]]
234 auto callDynamic(void* obj, std::span<const std::any> args) const -> std::any {
235 if (!invoke_dynamic) {
236 TOAST_WARN("Reflect", "callDynamic(): no dynamic invoker for '{}' (was it compiled with an older generator?)", name);
237 return {};
238 }
239 return invoke_dynamic(obj, args);
240 }

326 static auto get(void* obj) -> std::any {
327 if constexpr (std::is_enum_v<FieldType>) {
328 using Underlying = std::underlying_type_t<FieldType>;

331 return std::any {static_cast<int64_t>(value)};
