39constexpr auto quantize(float value, float steps_per_unit, float max_steps) -> float {
40 if (not(value > 0.0f)) {
41 return 0.0f;
42 }
43 const float steps = (value * steps_per_unit) + 0.5f;
44 return steps >= max_steps ? max_steps : static_cast<float>(static_cast<uint32_t>(steps));
