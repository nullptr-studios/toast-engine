56template<typename Callback>
57void queryVoxelSurface(const VoxelQueryContext& context, const AABB& aabb, Callback&& callback) {
58 // floor and ceil before clamping or a negative corner loses the info needed to reject it
59 glm::ivec3 first = glm::ivec3(glm::floor(aabb.min / voxel::k_voxel_size));
60 glm::ivec3 last = glm::ivec3(glm::ceil(aabb.max / voxel::k_voxel_size)) - 1;
61
62 const glm::ivec3 voxel_dims = glm::ivec3(context.volume.voxelDims());
63 if (glm::any(glm::lessThan(last, glm::ivec3(0))) || glm::any(glm::greaterThanEqual(first, voxel_dims))) {
64 return;
65 }
66 first = glm::clamp(first, glm::ivec3(0), voxel_dims - 1);
67 last = glm::clamp(last, glm::ivec3(0), voxel_dims - 1);
68 if (first.x > last.x || first.y > last.y || first.z > last.z) {
69 return;
70 }
71
72 const glm::ivec3 first_brick = first >> 3;
73 const glm::ivec3 last_brick = last >> 3;
74 const glm::ivec3 brick_dims = glm::ivec3(context.volume.brickDims());
75 const auto brick_dim = static_cast<int32_t>(voxel::k_brick_dim);
76
77 bool stopped = false;
78
79 const auto process_brick = [&](glm::ivec3 brick) {
80 const voxel::BrickEntry entry = context.volume.entryAt(brick);
81 if (entry.tag() == voxel::BrickTag::empty) {
82 return;
83 }
84
85 const std::span<const voxel::SurfaceVoxel> surface = context.surface.brickSurface(brick);
86 if (surface.empty()) {
87 return;
88 }
89
90 // private in Volume and VolumeSurface so recompute it the same way here
91 const uint32_t brick_slot = static_cast<uint32_t>(brick.x) + (static_cast<uint32_t>(brick.y) * brick_dims.x) +
92 (static_cast<uint32_t>(brick.z) * brick_dims.x * brick_dims.y);
93
94 const glm::ivec3 brick_base = brick * brick_dim;
95 const auto emit = [&](const voxel::SurfaceVoxel& surface_voxel, glm::ivec3 global_voxel) -> bool {
96 const uint8_t palette_index = context.volume.materialAt(global_voxel);
97 const uint32_t material_index = voxel::resolveMaterialIndex(context.palette, context.materials, palette_index);
98 const voxel::PhysicalMaterial& material = context.materials.materials[material_index];
99 if (not material.collides()) {
100 return true;
101 }
102
103 const glm::vec3 min = glm::vec3(global_voxel) * voxel::k_voxel_size;
104 const glm::vec3 max = min + glm::vec3(voxel::k_voxel_size);
105
106 return _detail::invokeVoxelCallback(
107 callback,
108 VoxelCandidate {
109 .voxel = global_voxel,
110 .brick = brick,
111 .brick_slot = brick_slot,
112 .local_index = surface_voxel.local_index,
113 .min = min,
114 .max = max,
115 .normal_index = voxel::normalOf(surface_voxel.classification),
116 .classification = voxel::classOf(surface_voxel.classification),
117 .palette_index = palette_index,
118 .material = &material,
119 }
120 );
121 };
122
123 const glm::ivec3 clipped_min = glm::max(first, brick_base);
124 const glm::ivec3 clipped_max = glm::min(last, brick_base + (brick_dim - 1));
125 const glm::ivec3 clipped_dims = clipped_max - clipped_min + 1;
126 const auto clipped_voxels =
127 static_cast<size_t>(clipped_dims.x) * static_cast<size_t>(clipped_dims.y) * static_cast<size_t>(clipped_dims.z);
128
129 if (clipped_voxels >= surface.size()) {
130 for (const voxel::SurfaceVoxel& surface_voxel : surface) {
131 const voxel::BrickCoord local = voxel::localFromIndex(surface_voxel.local_index);
132 const glm::ivec3 global_voxel =
133 brick_base + glm::ivec3 {static_cast<int32_t>(local.x), static_cast<int32_t>(local.y), static_cast<int32_t>(local.z)};
134
135 if (glm::any(glm::lessThan(global_voxel, first)) || glm::any(glm::greaterThan(global_voxel, last))) {
136 continue;
137 }
138 if (not emit(surface_voxel, global_voxel)) {
139 stopped = true;
140 return;
141 }
142 }
143 return;
144 }
145
146 for (int32_t vz = clipped_min.z; vz <= clipped_max.z; ++vz) {
147 for (int32_t vy = clipped_min.y; vy <= clipped_max.y; ++vy) {
148 for (int32_t vx = clipped_min.x; vx <= clipped_max.x; ++vx) {
149 const auto local_index = static_cast<uint16_t>(
150 (static_cast<uint32_t>(vz - brick_base.z) << 6u) | (static_cast<uint32_t>(vy - brick_base.y) << 3u) |
151 static_cast<uint32_t>(vx - brick_base.x)
152 );
153
154 const auto found = std::lower_bound(
155 surface.begin(), surface.end(), local_index, [](const voxel::SurfaceVoxel& candidate, uint16_t index) {
156 return candidate.local_index < index;
157 }
158 );
159 if (found == surface.end() || found->local_index != local_index) {
160 continue;
161 }
162 if (not emit(*found, glm::ivec3 {vx, vy, vz})) {
163 stopped = true;
164 return;
165 }
166 }
167 }
168 }
169 };
170
171 const uint64_t box_bricks = static_cast<uint64_t>(last_brick.x - first_brick.x + 1) *
172 static_cast<uint64_t>(last_brick.y - first_brick.y + 1) *
173 static_cast<uint64_t>(last_brick.z - first_brick.z + 1);
174 const uint64_t populated_bricks = static_cast<uint64_t>(context.surface.populatedBrickCount());
175
176 if (populated_bricks < box_bricks) {
177 context.surface.forEachPopulatedBrick([&](glm::ivec3 brick) {
178 if (stopped) {
179 return;
180 }
181 if (glm::any(glm::lessThan(brick, first_brick)) || glm::any(glm::greaterThan(brick, last_brick))) {
182 return;
183 }
184 process_brick(brick);
185 });
186 } else {
187 for (int32_t z = first_brick.z; z <= last_brick.z && not stopped; ++z) {
188 for (int32_t y = first_brick.y; y <= last_brick.y && not stopped; ++y) {
189 for (int32_t x = first_brick.x; x <= last_brick.x && not stopped; ++x) {
190 process_brick({x, y, z});
191 }
192 }
193 }
194 }
195}
